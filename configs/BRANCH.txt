! BRANCH - Router (VPN peer). Same securityk9 license step as EDGE.
enable
configure terminal
hostname BRANCH
no ip domain-lookup
interface gi0/0
 description TO-ISP-Gi0/1
 ip address 200.0.1.2 255.255.255.252
 no shutdown
interface gi0/1
 description BRANCH-LAN
 ip address 192.168.50.1 255.255.255.0
 no shutdown
ip route 0.0.0.0 0.0.0.0 200.0.1.1
access-list 101 permit ip 192.168.50.0 0.0.0.255 10.1.0.0 0.0.255.255

crypto isakmp policy 10
 encryption aes
 hash sha
 authentication pre-share
 group 2
 lifetime 86400
crypto isakmp key VPNkey123 address 200.0.0.1
crypto ipsec transform-set TS esp-aes esp-sha-hmac
crypto map CMAP 10 ipsec-isakmp
 set peer 200.0.0.1
 set transform-set TS
 match address 101
interface gi0/0
 crypto map CMAP
end
write memory
